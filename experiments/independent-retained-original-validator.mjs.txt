import fs from 'node:fs/promises';import crypto from 'node:crypto';
import {runRain,makeExample} from './hamon-a755e25-earth-noisy-inverse/core.js';
import {hypothesisLayer} from './hamon-a755e25-earth-noisy-inverse/draining-layer-evidence.js';
const path='/tmp/hamon-a755e25-second-independent-water-cases.json',bytes=await fs.readFile(path),source=JSON.parse(bytes),results=[];
for(const c of source.cases){const project=makeExample(),cells=Array.from({length:4},()=>({elevationM:0,surface:'garden',provenance:'Independent authored control; not field measurement'}));Object.assign(project.weather,{rainMmHr:c.rainMmHr,stormMinutes:c.stormMinutes,durationMinutes:Math.max(c.firstMinute,c.secondMinute)+60,stepMinutes:5,cellSizeM:1,mobilityPerHour:0,boundaryExitPerHour:0});const hypothesis={capacityMm:c.capacityMm,initialStoredMm:c.capacityMm*c.initialStoredFraction,drainagePerHour:c.drainagePerHour,source:'Frozen independent parameter tuple; not fitted'};const run=runRain(cells,2,2,project.weather,{drainingLayer:hypothesisLayer({surface:'garden'},hypothesis)});const depth=t=>run.snapshots.find(x=>x.minute===t).depthsMm[0],water=t=>run.drainingLayer.snapshots.find(x=>x.minute===t).retainedLiters,storm=depth(c.stormMinutes),first=water(c.firstMinute),second=water(c.secondMinute),reasons=[];if(!(c.firstMinute<c.secondMinute))reasons.push('late pair is not increasing in time');if(!(storm>1e-9))reasons.push('no positive storm-end surface depth');if(depth(c.firstMinute)>1e-9||depth(c.secondMinute)>1e-9)reasons.push('late surface is not dry');if(!(first>second&&second>0))reasons.push('truth retained pair is not positive decreasing');const midpoint=(first+second)/2,uncertainty=Math.abs(first-second)/2+.011,recorded=[midpoint-.01,midpoint+.01];if(recorded[0]<0)reasons.push('fixed perturbation gives negative retained center');results.push({id:c.id,eligible:!reasons.length,reasons,truth:{stormDepthMm:storm,firstDepthMm:depth(c.firstMinute),secondDepthMm:depth(c.secondMinute),firstRetainedLiters:first,secondRetainedLiters:second,initialStoredMm:hypothesis.initialStoredMm},recorded:{firstRetainedLiters:recorded[0],secondRetainedLiters:recorded[1],uncertaintyLiters:uncertainty},truthWithinBands:Math.abs(first-recorded[0])<=uncertainty&&Math.abs(second-recorded[1])<=uncertainty});}
const out={sourceSha256:crypto.createHash('sha256').update(bytes).digest('hex'),method:'Unchanged maintained runRain only; no root new inverse run, no parameter or perturbation retuning',eligible:results.filter(x=>x.eligible).length,total:results.length,results};await fs.writeFile('/tmp/hamon-a755e25-second-independent-water-validation.json',JSON.stringify(out,null,2));console.log(JSON.stringify({eligible:out.eligible,total:out.total,excluded:results.filter(x=>!x.eligible).map(x=>({id:x.id,reasons:x.reasons})),allTruthInsideDeclaredBands:results.every(x=>x.truthWithinBands)}));
